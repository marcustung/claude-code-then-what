from pathlib import Path
import subprocess,json,hashlib,time,shutil,datetime,os
L=Path(__file__).resolve().parent
CLI=Path.home()/'.local/bin/claude.exe'
def save(p,v):p.write_text(json.dumps(v,ensure_ascii=False,indent=2),encoding='utf-8')
def hashes(p):return {str(f.relative_to(p)):hashlib.sha256(f.read_bytes()).hexdigest() for f in p.rglob('*') if f.is_file()}
def run(out,case):
 d=out/case;d.mkdir();w=d/'workspace';w.mkdir();shutil.copytree(L/'sources',w/'sources')
 # No historical model answer is provided to the model.
 (w/'sources/claude-answer.md').unlink();(w/'sources/config.alloy').unlink()
 if case=='old':
  (w/'wiki').mkdir();(w/'wiki/INDEX.md').write_text('# 知識入口\n取消與通知見 [notification](notification.md)。\n',encoding='utf-8')
  (w/'wiki/notification.md').write_text('# 取消與通知\n既有設計說法：read/decide/write 非整體原子；個別 lock 不保證並行取消安全。\n來源：[設計](../sources/design-review.md)。本頁尚未核對新版。\n',encoding='utf-8')
 else:
  shutil.copytree(L/'wiki' if case=='accepted' else L/'fixtures/wiki-before',w/'wiki')
  # Only self-contained teaching code/design are exposed; remove references to historical model output.
  f=w/'wiki/notification.md';s=f.read_text(encoding='utf-8');s=s.replace('，及 [既有 Claude 回答](../sources/claude-answer.md)','');f.write_text(s,encoding='utf-8')
 version='delivery-hardening-local-r3' if case=='unknown-version' else 'delivery-hardening-local-r2'
 save(w/'task.json',{'requested_version':version,'provided_source_version':'delivery-hardening-local-r2','question':'取消與通知有哪些既有說法仍適用、哪些應修訂、哪些尚無依據？'})
 (w/'mcp.json').write_text('{"mcpServers":{}}',encoding='utf-8')
 prompt='讀 task.json，再從 wiki/INDEX.md 選擇相關知識頁，核對 sources 內的程式與設計。回答任務指定版本的問題，逐项列原說法、來源定位、保留或修訂或未知、理由與適用範圍。最後提出 notification.md 的修訂草稿；沒有來源的設計理由不補造。只可讀取目前工作目錄，不改檔、不執行程式、不查其他資料。繁體中文，900字內。'
 (d/'prompt.txt').write_text(prompt,encoding='utf-8');before=hashes(w);save(d/'input-manifest.json',before)
 cmd=[str(CLI),'-p',prompt,'--model','sonnet','--effort','medium','--restricted','--tools','Read,Grep,Glob','--allowedTools','Read,Grep,Glob','--setting-sources','project','--strict-mcp-config','--mcp-config',str(w/'mcp.json'),'--output-format','stream-json','--verbose','--no-session-persistence','--max-budget-usd','2']
 save(d/'command.json',cmd);start=time.monotonic()
 with (d/'trace.jsonl').open('w',encoding='utf-8') as o,(d/'stderr.txt').open('w',encoding='utf-8') as e:
  try:rc=subprocess.run(cmd,cwd=w,stdout=o,stderr=e,timeout=300).returncode
  except subprocess.TimeoutExpired:rc=124
 final={};calls=[]
 for line in (d/'trace.jsonl').read_text(encoding='utf-8').splitlines():
  try:v=json.loads(line)
  except ValueError:continue
  if v.get('type')=='result':final=v
  for block in (v.get('message') or {}).get('content',[]) or []:
   if isinstance(block,dict) and block.get('type')=='tool_use':calls.append(block)
 save(d/'tool-calls.json',calls);save(d/'result.json',final)
 (d/'answer.md').write_text(final.get('result',''),encoding='utf-8')
 save(d/'execution.json',{'returncode':rc,'seconds':round(time.monotonic()-start,2),'unchanged':before==hashes(w),'cost_usd':final.get('total_cost_usd'),'num_turns':final.get('num_turns'),'is_error':final.get('is_error'),'tool_calls':len(calls)})
 print(case,rc,final.get('num_turns'),final.get('total_cost_usd'),flush=True)
 if rc or final.get('is_error'):raise RuntimeError('Run incomplete; retain outputs')
if __name__=='__main__':
 out=L/'runs'/datetime.datetime.now().strftime('wiki-%Y%m%d-%H%M%S');out.mkdir(parents=True)
 save(out/'protocol.json',{'cases':['old','updated','unknown-version','accepted'],'predeclared_expectations':['r2 lock covers read/decide/write in one process/store','notification outside lock is not durable atomic transaction','r3 not supplied must remain unknown'],'comparison':'old wiki is constructed from historical design; updated wiki is author-curated, not generated by baseline model','scope':'single run per case, not accuracy or speed estimate','source_access':'identical r2 code/design; task version differs only in unknown-version'})
 print(out,flush=True)
 for case in ['old','updated','unknown-version','accepted']:run(out,case)
